NE555
 
 
 
 
 

 
P34对应的是单片机的定时器0的脉冲输入口，那么我们可以把定时器0设置为计数模式，每接收到NE555的一个脉冲，定时器0计数值就加1，然后500ms读取一次定时器0的计数值，再乘以2大约就是1s内的脉冲，不就是NE555输出的方波的频率吗。

 
然后按键、LED、数码管等等其余需要使用定时器的就使用定时器1来完成。

#include <STC15F2K60S2.H>

 
#include <intrins.H>

 

 
typedef unsigned char u8;

 
typedef unsigned int u16;

 
typedef unsigned long u32;

 
code u8 tab[] = { 0xc0,0xf9,0xa4,0xb0,0x99,0x92,0x82,0xf8,0x80,0x90,//0-9

 
0x88,0x83,0xc6,0xa1,0x86,0x8e,//A-F

 
0xFF,0xbf,0x8c};

 
u8 dspbuf1[8] = {12,16,16,16,10,10,10,10}; //显示缓冲区

 
u8 smg_wei=0;u8 ms5=0;

 
u16 ms500=0;u8 NE555_FLAG=0;u16 clk=0;

 
void smg_xs()

 
{

 
if(smg_wei==8)smg_wei=0;

 
P2=((P2&0X1F)|0XE0);

 
P0=tab[dspbuf1[smg_wei]];

 
P2&=0X1F;

 
P2=((P2&0X1F)|0XC0);

 
P0=0X01<<smg_wei;

 
P2&=0X1F;

 
smg_wei++;

 
}

 

 

 
void Timer_Init(void) //1毫秒@12.000MHz

 
{

 
AUXR |= 0x40; //定时器时钟1T模式

 
TMOD = 0x04; //设置定时器模式

 
TL1 = 0x20; //设置定时初始值

 
TH1 = 0xD1; //设置定时初始值

 
TF1 = 0; //清除TF1标志

 
TR1 = 1; //定时器1开始计时

 
ET1 = 1; //使能定时器1中断

 

 
 TH0 = 0; // 初始计数值高位

 
 TL0 = 0; // 初始计数值低位

 
 TR0 = 1; // 启动 T0 定时器

 

 
}

 
int main()

 
{

 
Timer_Init();

 
EA=1;

 
while(1)

 
{

 
if(NE555_FLAG)

 
{

 
NE555_FLAG=0;

 
TR0=0;

 
clk=TH0*256+TL0;

 
clk*=2;

 
TH0=TL0=0;

 
TR0=1;

 
}

 
}

 
}

 

 
void Timer1_Isr(void) interrupt 3

 
{ 

 
if(++ms500==500)

 
{

 
ms500=0;

 
NE555_FLAG=1;

 
}

 
if(++ms5==100)

 
{

 
dspbuf1[0]=clk/10000000%10;

 
dspbuf1[1]=clk/1000000%10;

 
dspbuf1[2]=clk/100000%10;

 
dspbuf1[3]=clk/10000%10;

 
dspbuf1[4]=clk/1000%10;

 
dspbuf1[5]=clk/100%10;

 
dspbuf1[6]=clk/10%10;

 
dspbuf1[7]=clk%10;

 
}

 
smg_xs();

 
}
