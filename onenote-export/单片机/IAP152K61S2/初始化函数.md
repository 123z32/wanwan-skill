初始化函数
 
 
 
 
 

 
//初始化函数，关闭蜂鸣器继电器、led灯

 
void sys_init()

 
{

 
 P2=P2&0X1F|0XA0; //打开Y5C

 
 P0=0X00; //关闭蜂鸣器继电器

 
 P2=P2&0X1F; //关闭Y5C

 

 
 P2=P2&0X1F|0X80; //打开Y4C

 
 P0=0Xff; //关闭led灯

 
 P2=P2&0X1F; //关闭Y4C

 
}
