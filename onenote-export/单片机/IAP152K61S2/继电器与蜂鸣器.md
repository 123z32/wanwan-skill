#include <public.H>

 
u16 SP_TIME=0;

 
u16 SP_TIME1=0;

 
void Timer1_Init(void) //1毫秒@12.000MHz

 
{

 
AUXR |= 0x40; //定时器时钟1T模式

 
TMOD &= 0x0F; //设置定时器模式

 
TL1 = 0x20; //设置定时初始值

 
TH1 = 0xD1; //设置定时初始值

 
TF1 = 0; //清除TF1标志

 
TR1 = 1; //定时器1开始计时

 
ET1 = 1; //使能定时器1中断

 
}

 
void SP_K(bit SP,bit K)

 
{

 
//继电器P04 P04=1;开 P04=0;关

 
//蜂鸣器P06 P06=1;开 P06=0;关

 
P2=P2&0X1F|0XA0;

 
P06=SP;P04=K;

 
 P2=P2&0X1F; 

 
}

 

 
int main()

 
{

 
Timer1_Init();

 
EA=1;

 
while(1)

 
{

 

 
}

 
}

 

 
void Timer1_Isr(void) interrupt 3

 
{

 
if(++SP_TIME==200)

 
{

 
SP_TIME1=0;

 
SP_K(0,1);

 
}

 
if(++SP_TIME1==500)

 
{

 
SP_TIME=0;

 
SP_K(1,0);

 
}

 
}
